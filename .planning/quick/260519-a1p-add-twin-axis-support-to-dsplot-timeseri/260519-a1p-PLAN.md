import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from dsplot import TimeSeriesPanel, TimeSeries, Line, style, Figure

panel = TimeSeriesPanel(
    title='twin test',
    units=(3, 1),
    twin_y=True,
    twin_y_label='f (Hz)',
    twin_y_side='left',
    twin_yticks=[0.0, 0.5, 1.0],
    twin_ytick_labels=['200', '2k', '20k'],
    twin_ylim=(0.0, 1.0),
)
panel.add(TimeSeries(np.sin(np.linspace(0, 10, 1000)), 44100))
panel.add_twin(Line(np.linspace(0, 1, 100), np.linspace(0, 1, 100)))
fig = Figure.compose(rows=[[panel]])
fig.render()
assert panel.ax_twin is not None, 'twin axis not created'
assert panel.ax_twin.yaxis.get_label().get_text() == 'f (Hz)'
ticks = [t.get_text() for t in panel.ax_twin.get_yticklabels()]
assert ticks == ['200', '2k', '20k'], f'unexpected tick labels: {ticks}'
ylim = panel.ax_twin.get_ylim()
assert ylim == (0.0, 1.0), f'unexpected ylim: {ylim}'
print('OK')
plt.close('all')
