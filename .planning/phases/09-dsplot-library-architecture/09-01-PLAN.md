```python
# research/dsplot/plottables/base.py
from __future__ import annotations
from abc import ABC, abstractmethod
from matplotlib.axes import Axes

class Plottable(ABC):
    """A unit drawn onto a matplotlib Axes.

    Style knobs are settable at construction (kwargs) and mutable after
    (attribute assignment). draw(ax) is called by the owning Panel.
    None-valued knobs resolve lazily against dsplot.style.* at draw() time —
    this is what makes runtime style reassignment work (per D-05 lazy lookup).
    """
    color: str | None
    linewidth: float | None
    alpha: float
    linestyle: str
    label: str | None
    zorder: int

    @abstractmethod
    def draw(self, ax: Axes) -> None:
        """Render this plottable onto ax. Pure side-effect on ax; returns None."""
        ...
```

```python
# research/dsplot/plottables/vector.py
# Polymorphic Vector — accepts 2-tuple OR 3-tuple input per D-02.
# 2-tuple → 2D rendering on a regular mpl Axes (FancyArrowPatch shaft + head)
# 3-tuple → 3D rendering on an Axes3D (ax.plot + scatter tip + 3D text label)
# Same API surface; the class branches internally on len(vec) at draw() time.

class Vector(Plottable):
    """Single arrow from origin to origin + vec.

    Dimensionality inferred from len(vec):
      - 2 → 2D (requires a regular mpl Axes)
      - 3 → 3D (requires an Axes3D; e.g. axes created with projection='3d')

    Origin defaults match dimensionality automatically when not provided.
    """
    def __init__(self,
                 vec: tuple[float, float] | tuple[float, float, float],
                 *,
                 origin: tuple[float, float] | tuple[float, float, float] | None = None,
                 color: str | None = None,
                 linewidth: float | None = None,
                 alpha: float = 1.0,
                 linestyle: str = "-",
                 label: str | None = None,
                 label_offset: float | tuple[float, ...] | None = None,
                 show_tip: bool = True,         # only meaningful for 3D path (scatter at tip)
                 zorder: int = 2): ...
```
