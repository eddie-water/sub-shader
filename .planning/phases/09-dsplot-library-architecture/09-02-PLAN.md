# research/dsplot/panels/base.py
from __future__ import annotations
from abc import ABC, abstractmethod
from matplotlib.axes import Axes
from ..plottables.base import Plottable

class Panel(ABC):
    """Container that owns one mpl Axes and a list of Plottables.

    Subclasses (StaticPanel / DynamicPanel / InteractivePanel) differ only in
    how/when they call render() — the composition contract is shared.
    """
    ax: Axes | None              # set by Figure.attach() at layout time
    _plottables: list[Plottable]

    def add(self, plottable: Plottable) -> "Panel":
        """Append plottable to this panel's render list. Returns self for chaining."""
        ...

    def attach(self, ax: Axes) -> None:
        """Called by Figure when this panel is placed into a gridspec cell."""
        ...

    @abstractmethod
    def render(self) -> None:
        """Draw all plottables onto self.ax. Subclasses override for animation/interactivity."""
        ...
```

# research/dsplot/figure.py
class Figure:
    """Orchestrator that composes Panels into one mpl Figure.

    Layout defaults resolve against dsplot.style.DEFAULT_* per D-05 (figsize from
    DEFAULT_PANEL_SIZE_INCHES, hspace from DEFAULT_HSPACE, etc.) — Figure NEVER
    hardcodes layout numerics.

    Per D-04, add_panel accepts a `projection` kwarg so 3D cells (Axes3D) compose
    via the same pattern as 2D cells:
        fig.add_panel(panel_3d, row=0, col=0, projection="3d")

    Usage:
        fig = Figure(n_rows=1, n_cols=3)   # figsize inherited from style.DEFAULT_PANEL_SIZE_INCHES
        fig.add_panel(StaticPanel(title="Panel A"), row=0, col=0)
        fig.add_panel(StaticPanel(title="Panel B"), row=0, col=1)
        fig.add_panel(StaticPanel(title="Panel C"), row=0, col=2)
        fig.render()
        fig.savefig("out.png")
        fig.show()         # interactive / notebook contexts
    """
    def __init__(self, *,
                 n_rows: int = 1, n_cols: int = 1,
                 figsize: tuple[float, float] | None = None,
                 suptitle: str | None = None,
                 hspace: float | None = None,
                 wspace: float | None = None,
                 width_ratios: list[float] | None = None,
                 height_ratios: list[float] | None = None,
                 dpi: int | None = None): ...

    def add_panel(self, panel: Panel, *, row: int = 0, col: int = 0,
                  rowspan: int = 1, colspan: int = 1,
                  projection: str | None = None) -> Panel:
        """Attach a Panel to gridspec[row:row+rowspan, col:col+colspan].

        projection="3d" → Axes3D for this cell (per D-04).
        projection=None → regular 2D Axes (default).
        """
        ...

    def render(self) -> None: ...
    def savefig(self, path: str, **kwargs) -> str: ...
    def show(self) -> None: ...
    def close(self) -> None: ...
```

<risks>
- **Render-time vs construct-time axis attachment:** Decision: Figure.render() attaches axes to panels. If a consumer calls panel.render() without going through Figure first, they hit the "render() called before attach()" RuntimeError. That's the intended contract; document it in StaticPanel.render's docstring. Wave 2's DynamicPanel needs to play by the same rule (FuncAnimation orchestration happens AFTER attach).
- **Default layout opinionation (now driven by D-05 template):** Figure picks defaults from `style.DEFAULT_HSPACE` (0.18) and `style.DEFAULT_WSPACE` (0.04). If a Wave 3 figure port needs different spacing, it passes a per-instance `hspace=...` kwarg, OR (per D-05 local-derive pattern) defines a module-level local constant: `from dsplot import style; HSPACE = style.DEFAULT_HSPACE * 1.5`.
