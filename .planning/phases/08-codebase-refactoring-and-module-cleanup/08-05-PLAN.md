       ```python
       """SubShader pipeline — audio visualization orchestrator."""

       from subshader.utils.logging import get_logger
       from subshader.utils.gpu import gpu_available
       from subshader.audio import AudioStream
       from subshader.dsp.cwt import GpuCWT, CpuCWT
       from subshader.renderer import Renderer

       log = get_logger(__name__)

       class SubShader:
           """Main pipeline: AudioStream -> CWT -> Renderer."""

           def __init__(self, config):
               self.config = config
               self.audio = AudioStream(config)       # discovers sample_rate from file
               self.dsp = GpuCWT(config) if gpu_available() else CpuCWT(config)
               self.renderer = Renderer(config, self.dsp.output_shape)

           def run(self):
               """Main visualization loop."""
               self.audio.start()
               while not self.renderer.should_close():
                   chunk = self.audio.next_chunk()
                   coefs = self.dsp.process(chunk)
                   self.renderer.update(coefs)

           def cleanup(self):
               """Release all resources. Safe to call multiple times."""
               if hasattr(self, 'audio') and self.audio:
                   self.audio.cleanup()
               if hasattr(self, 'dsp') and self.dsp:
                   self.dsp.cleanup()
               if hasattr(self, 'renderer') and self.renderer:
                   self.renderer.cleanup()
       ```

       Per D-20: The orchestrator sees three modules and a config. No knowledge of readers, players, kernels, frame buffers, or shaders. The run() method should look like pseudocode.

       Preserve the existing exception handling pattern from current __main__.py — catch GRACEFUL_EXCEPTIONS in run() or let them propagate to __main__.py's try/except.

    2. Rewrite `src/subshader/__main__.py` to ~15 lines per D-18:
       ```python
       #!/usr/bin/env python3
       """SubShader — real-time audio visualization."""

       import argparse
       from subshader.utils.logging import logger_init
       from subshader.config import CWTConfig
       from subshader.pipeline import SubShader
       from subshader.exceptions import GRACEFUL_EXCEPTIONS, ExceptionReporter

       def parse_args():
           parser = argparse.ArgumentParser(description="SubShader audio visualizer")
           parser.add_argument("audio_file", nargs="?", default=None,
                               help="Path to audio file (WAV)")
           return parser.parse_args()

       def main():
           logger_init(log_level="INFO", console_output=True, file_output=True)
           args = parse_args()

           config = CWTConfig()
           if args.audio_file:
               config.file_path = args.audio_file

           pipeline = SubShader(config)
           try:
               pipeline.run()
           except GRACEFUL_EXCEPTIONS as e:
               ExceptionReporter.report(e)
           finally:
               pipeline.cleanup()

       if __name__ == "__main__":
           main()
       ```

       Check existing __main__.py for any other CLI args or config overrides that must be preserved (e.g., the current code modifies config.audio.file_path). Migrate those to the new pattern.
